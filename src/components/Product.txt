import React, { useState, useEffect, useRef } from "react";
import { useNavigate, useParams } from "react-router-dom";
import axios from "axios";
import { generateProductDescription, generateSpeech } from "../utils/aiServices";
import { motion } from "framer-motion";
import { Toaster, toast } from "react-hot-toast";

const ProductDetail = ({ containerStyle, addToCart }) => {
	const navigate = useNavigate();
	const { id } = useParams();
	const [product, setProduct] = useState(null);
	const [loading, setLoading] = useState(true);
	const [error, setError] = useState(null);
	const [generatedDescription, setGeneratedDescription] = useState(null);
	const [isGenerating, setIsGenerating] = useState(false);
	const [isReading, setIsReading] = useState(false);
	const [quantity, setQuantity] = useState(1);
	const [expanded, setExpanded] = useState(false);
	const [similar, setSimilar] = useState([]);
	const mountedRef = useRef(true);

	useEffect(() => {
		mountedRef.current = true;
		return () => {
			mountedRef.current = false;
		};
	}, []);

	useEffect(() => {
		const fetchProduct = async () => {
			setLoading(true);
			try {
				const res = await axios.get(`https://dummyjson.com/products/${id}`);
				if (!mountedRef.current) return;
				setProduct(res.data);
			} catch (err) {
				if (!mountedRef.current) return;
				setError(err.message || "Failed to load product.");
			} finally {
				if (!mountedRef.current) return;
				setLoading(false);
			}
		};
		fetchProduct();
	}, [id]);

	useEffect(() => {
		const fetchSimilar = async () => {
			if (!product?.category) return;
			try {
				const res = await axios.get(
					`https://dummyjson.com/products/category/${encodeURIComponent(product.category)}?limit=6`
				);
				const items = res.data?.products?.filter((p) => String(p.id) !== String(product.id)) || [];
				setSimilar(items);
			} catch (_) {
				// silent
			}
		};
		fetchSimilar();
	}, [product]);

	const handleGenerateDescription = async () => {
		if (!product) return;
		setIsGenerating(true);
		setError(null);
		try {
			const desc = await generateProductDescription(product);
			setGeneratedDescription(desc);
		} catch (err) {
			setError(err?.message || "Failed to generate description.");
			toast.error("AI description failed");
		} finally {
			setIsGenerating(false);
		}
	};

	const handleReadDescription = async () => {
		const text = generatedDescription || product?.description;
		if (!text) return;
		setIsReading(true);
		setError(null);
		try {
			// Try app-provided AI TTS first
			if (generateSpeech) {
				await generateSpeech(text);
			} else if (typeof window !== "undefined" && window.speechSynthesis) {
				// Fallback to browser TTS
				const utter = new SpeechSynthesisUtterance(text);
				window.speechSynthesis.cancel();
				window.speechSynthesis.speak(utter);
			} else {
				throw new Error("No TTS available");
			}
		} catch (err) {
			setError(err?.message || "Failed to play audio");
			toast.error("TTS failed");
		} finally {
			setIsReading(false);
		}
	};

	const handleCopyDescription = async () => {
		try {
			const text = generatedDescription || product?.description || "";
			await navigator.clipboard.writeText(text);
			toast.success("Description copied to clipboard");
		} catch (err) {
			toast.error("Copy failed");
		}
	};

	const inc = () => setQuantity((q) => Math.min(99, q + 1));
	const dec = () => setQuantity((q) => Math.max(1, q - 1));

	if (loading) return <div className="text-center pt-32 text-2xl font-semibold">Loading product details...</div>;
	if (error && !product) return <div className="text-center pt-32 text-red-500 text-2xl font-semibold">Error loading product: {error}</div>;
	if (!product) return <div className="text-center pt-32 text-2xl font-semibold">Product not found.</div>;

	return (
		<div className="container mx-auto p-4 md:p-8 pt-32 min-h-screen" style={containerStyle}>
			<Toaster position="top-center" />
			<div className="bg-white dark:bg-slate-900 rounded-xl shadow-lg p-6 md:p-12 flex flex-col md:flex-row gap-8">
				<motion.div whileHover={{ scale: 0.98 }} className="md:w-1/2 flex justify-center items-center">
					<img
						src={product.thumbnail || "https://placehold.co/400x400/7c3aed/ffffff?text=Image+Not+Found"}
						alt={product.title}
						className="rounded-3xl w-full max-w-md h-auto object-cover bg-center shadow-inner"
						onError={(e) => { e.target.src = "https://placehold.co/400x400/7c3aed/ffffff?text=Image+Not+Found"; }}
					/>
				</motion.div>

				<div className="md:w-1/2 flex flex-col justify-center">
					<h1 className="text-3xl md:text-4xl font-bold text-violet-800 mb-2">{product.title}</h1>
					<p className="text-sm md:text-lg text-gray-600 mb-2 capitalize">{product.category}</p>
					<p className="text-2xl md:text-3xl font-extrabold text-gray-900 mb-4">${Number(product.price).toFixed(2)}</p>

					<div className="flex flex-wrap gap-3 mb-4">
						<button
							onClick={handleGenerateDescription}
							disabled={isGenerating}
							aria-disabled={isGenerating}
							className="px-4 py-2 bg-violet-600 text-white rounded-md shadow-sm hover:bg-violet-700 disabled:opacity-60"
						>
							{isGenerating ? "Generating..." : "Generate AI Description ✨"}
						</button>

						<button
							onClick={handleReadDescription}
							disabled={isReading || (!generatedDescription && !product.description)}
							aria-disabled={isReading || (!generatedDescription && !product.description)}
							className="px-4 py-2 bg-indigo-600 text-white rounded-md shadow-sm hover:bg-indigo-700 disabled:opacity-60"
						>
							{isReading ? "Reading..." : "Read Description 🔊"}
						</button>

						<button onClick={handleCopyDescription} className="px-4 py-2 bg-gray-200 text-gray-800 rounded-md shadow-sm hover:bg-gray-300">Copy</button>
					</div>

					<div className="mb-4">
						<h3 className="text-lg font-semibold text-gray-800 mb-2">Description</h3>
						<div className="text-gray-700 text-base leading-relaxed bg-gray-50 p-4 rounded-md">
							<p className={expanded ? "" : "line-clamp-4"}>{generatedDescription || product.description}</p>
							{generatedDescription && <p className="text-sm text-green-600 mt-2 italic">✨ AI Generated Description</p>}
						</div>
						<button onClick={() => setExpanded((s) => !s)} className="mt-2 text-sm text-violet-700">{expanded ? "Show less" : "Show more"}</button>
					</div>

					<div className="flex items-center gap-4 mb-6">
						<div className="flex items-center border rounded-md">
							<button aria-label="decrease quantity" onClick={dec} type="button" className="px-3 py-2">-</button>
							<div className="px-4">{quantity}</div>
							<button aria-label="increase quantity" onClick={inc} type="button" className="px-3 py-2">+</button>
						</div>

						<button
							onClick={() => { addToCart({ ...product, quantity }); toast.success(`${product.title} x${quantity} added to cart`); }}
							className="px-5 py-2 bg-green-500 text-white rounded-md shadow hover:bg-green-600"
						>
							Add to Cart
						</button>

						<button onClick={() => navigate("/")} className="px-4 py-2 bg-gray-200 rounded-md">Go Back</button>
					</div>

					{similar.length > 0 && (
						<div className="mt-4">
							<h4 className="text-md font-semibold mb-2">More like this</h4>
							<div className="flex gap-3 overflow-x-auto py-2">
								{similar.map((p) => (
									<div key={p.id} className="min-w-[160px] bg-white border rounded-md p-2 shadow-sm">
										<img src={p.thumbnail} alt={p.title} className="w-full h-24 object-cover rounded" />
										<p className="text-sm font-medium mt-2 line-clamp-2">{p.title}</p>
										<p className="text-sm text-gray-600">${Number(p.price).toFixed(2)}</p>
										<button onClick={() => navigate(`/product/${p.id}`)} className="mt-2 w-full text-sm bg-violet-600 text-white rounded py-1">View</button>
									</div>
								))}
							</div>
						</div>
					)}

					{/* Reviews / Comments Section */}
					<div className="mt-6">
						<h4 className="text-lg font-semibold mb-3">Reviews & Feedback</h4>
						<ReviewForm
							productId={product.id}
							onAdd={(c) => {
								// reload comments after add
								const key = `comments_product_${product.id}`;
								const raw = localStorage.getItem(key);
								const list = raw ? JSON.parse(raw) : [];
								setComments(list);
							}}
						/>

						<ReviewList productId={product.id} onDelete={() => {
							const key = `comments_product_${product.id}`;
							const raw = localStorage.getItem(key);
							const list = raw ? JSON.parse(raw) : [];
							setComments(list);
						}} />
					</div>
				</div>
			</div>
		</div>
	);
};

// -- Reviews subcomponents (kept in-file for simplicity) --
const useComments = (productId) => {
	const key = `comments_product_${productId}`;
	const load = () => {
		try {
			const raw = localStorage.getItem(key);
			return raw ? JSON.parse(raw) : [];
		} catch {
			return [];
		}
	};
	const save = (items) => {
		try {
			localStorage.setItem(key, JSON.stringify(items));
		} catch {}
	};
	return { load, save };
};

const ReviewForm = ({ productId, onAdd }) => {
	const [name, setName] = useState("");
	const [text, setText] = useState("");
	const [rating, setRating] = useState(5);
	const [imagePreview, setImagePreview] = useState(null);
	const fileRef = useRef(null);
	const { load, save } = useComments(productId);

	useEffect(() => {
		// clear on mount
		setName("");
		setText("");
		setRating(5);
		setImagePreview(null);
	}, [productId]);

	const handleImage = (file) => {
		if (!file) return setImagePreview(null);
		const reader = new FileReader();
		reader.onload = () => setImagePreview(reader.result);
		reader.readAsDataURL(file);
	};

	const handleSubmit = (e) => {
		e.preventDefault();
		const item = {
			id: `${Date.now()}-${Math.random().toString(36).slice(2, 9)}`,
			name: name.trim() || "Anonymous",
			text: text.trim(),
			rating: Number(rating) || 0,
			image: imagePreview || null,
			createdAt: Date.now(),
		};
		const list = load();
		list.unshift(item);
		save(list);
		toast.success("Review added");
		setName("");
		setText("");
		setRating(5);
		setImagePreview(null);
		if (fileRef.current) fileRef.current.value = null;
		onAdd && onAdd(item);
	};

	return (
		<form onSubmit={handleSubmit} className="bg-gray-50 p-4 rounded-md mb-4">
			<div className="flex gap-3 flex-col md:flex-row">
				<input aria-label="Your name" value={name} onChange={(e) => setName(e.target.value)} placeholder="Your name" className="flex-1 p-2 border rounded" />
				<select aria-label="Rating" value={rating} onChange={(e) => setRating(e.target.value)} className="p-2 border rounded w-28">
					{[5,4,3,2,1].map((r) => (<option key={r} value={r}>{r} ★</option>))}
				</select>
			</div>
			<textarea aria-label="Your review" value={text} onChange={(e) => setText(e.target.value)} placeholder="Write your review" className="w-full mt-3 p-2 border rounded" rows={3} />

			<div className="flex items-center gap-3 mt-3">
				<input ref={fileRef} type="file" accept="image/*" onChange={(e) => handleImage(e.target.files?.[0])} />
				{imagePreview && <img src={imagePreview} alt="preview" className="w-20 h-20 object-cover rounded" />}
				<button type="submit" className="ml-auto px-4 py-2 bg-violet-600 text-white rounded">Add Review</button>
			</div>
		</form>
	);
};

const ReviewList = ({ productId, onDelete }) => {
	const { load, save } = useComments(productId);
	const [comments, setComments] = useState([]);

	useEffect(() => {
		setComments(load());
	}, [productId]);

	const handleDelete = (id) => {
		const next = comments.filter((c) => c.id !== id);
		save(next);
		setComments(next);
		toast.success("Review deleted");
		onDelete && onDelete(id);
	};

	if (!comments || comments.length === 0) return <p className="text-sm text-gray-500">No reviews yet — be the first!</p>;

	const avg = (comments.reduce((s, c) => s + (c.rating||0), 0) / comments.length).toFixed(1);

	return (
		<div className="space-y-3 mt-3">
			<div className="flex items-center gap-3">
				<strong>{avg} ★</strong>
				<span className="text-sm text-gray-600">({comments.length} reviews)</span>
			</div>
			{comments.map((c) => (
				<div key={c.id} className="border rounded p-3 bg-white">
					<div className="flex items-start gap-3">
						<div className="flex-1">
							<div className="flex items-center justify-between">
								<div className="font-semibold">{c.name}</div>
								<div className="text-sm text-gray-600">{new Date(c.createdAt).toLocaleString()}</div>
							</div>
							<div className="text-yellow-500">{'★'.repeat(c.rating)}{'☆'.repeat(5-c.rating)}</div>
							<p className="mt-2 text-sm text-gray-800">{c.text}</p>
							{c.image && <img src={c.image} alt="review" className="mt-2 w-32 h-32 object-cover rounded" />}
						</div>
						<button onClick={() => handleDelete(c.id)} className="text-sm text-red-600">Delete</button>
					</div>
				</div>
			))}
		</div>
	);
};

export default ProductDetail;
